const input = document.getElementById("question"); const button = document.getElementById("askButton"); const response = document.getElementById("response"); function askGoldenTouchAI() { const question = input.value.trim(); if (question === "") { response.textContent = "Please type a question first."; input.focus(); return; } button.disabled = true; button.textContent = "Thinking..."; response.textContent = "Golden Touch AI is thinking..."; setTimeout(function () { response.textContent = "Golden Touch AI is ready to help with your question: " + question; button.disabled = false; button.textContent = "Ask AI"; }, 1200); } button.addEventListener("click", askGoldenTouchAI); input.addEventListener("keydown", function (event) { if (event.key === "Enter") { event.preventDefault(); askGoldenTouchAI(); } });
